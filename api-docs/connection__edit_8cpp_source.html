358 void SetFixedParams() {
359 m_params.direction = PortDirection::kInOut;
360 m_params.network_address = "";
361 m_params.network_port = 0;
362 m_params.net_protocol = PROTO_UNDEFINED;
363 m_params.baudrate = 0;
364 }
365
366 void OnClick(wxCommandEvent&) {
367 if (m_bt_scanning)
368 StopScan();
369 else {
370 m_bt_no_change_counter = 0;
371 m_bt_last_result_count = 0;
372 m_scan_timer.Start(1000, wxTIMER_CONTINUOUS);
373 g_Platform->startBluetoothScan();
374 m_bt_scanning = 1;
375 if (m_scan_btn) {
376 m_scan_btn->SetLabel(_("Stop Scan"));
377 }
378 }
379 }
380
381 void OnTimer() {
382 if (!m_bt_scanning) return;
383 m_bt_scanning++;
384 m_bt_scan_results = g_Platform->getBluetoothScanResults();
385
386 m_src_choice->Clear();
387 m_src_choice->Append(m_bt_scan_results[0]); // scan status
388
389 unsigned i = 1;
390 while ((i + 1) < m_bt_scan_results.GetCount()) {
391 wxString item1 = m_bt_scan_results[i] + ";";
392 wxString item2 = m_bt_scan_results.Item(i + 1);
393 m_src_choice->Append(item1 + item2);
394 i += 2;
395 }
396
397 if (m_bt_scan_results.GetCount() > 1) {
398 m_src_choice->SetSelection(1);
399 }
400
401 // Watch for changes. When no changes occur after n seconds, stop the
402 // scan
403 if (m_bt_no_change_counter > 5) StopScan();
404 int count = static_cast<int>(m_bt_scan_results.GetCount());
405 if (count == m_bt_last_result_count)
406 m_bt_no_change_counter++;
407 else
408 m_bt_no_change_counter = 0;
409 m_bt_last_result_count = static_cast<int>(m_bt_scan_results.GetCount());
410
411 // Absolute fallback
412 if (m_bt_scanning >= 15) {
413 StopScan();
414 }
415 }
416
417 void StopScan() {
418 m_scan_timer.Stop();
419 g_Platform->stopBluetoothScan();
420 m_bt_scanning = 0;
421 if (m_scan_btn) {
422 m_scan_btn->SetLabel(_("BT Scan"));
423 m_scan_btn->Enable();
424 }
425 }
