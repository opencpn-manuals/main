359 void SetFixedParams() {
360 m_params.direction = PortDirection::kInOut;
361 m_params.network_address = "";
362 m_params.network_port = 0;
363 m_params.net_protocol = PROTO_UNDEFINED;
364 m_params.baudrate = 0;

367 void OnClick(wxCommandEvent&) {
368 if (m_bt_scanning)
369 StopScan();
370 else {
371 m_bt_no_change_counter = 0;
372 m_bt_last_result_count = 0;
373 m_scan_timer.Start(1000, wxTIMER_CONTINUOUS);
374 g_Platform->startBluetoothScan();
375 m_bt_scanning = 1;
376 if (m_scan_btn) {
377 m_scan_btn->SetLabel(_("Stop Scan"));
378 }
379 }
380 }
381
382 void OnTimer() {
383 if (!m_bt_scanning) return;
384 m_bt_scanning++;
385 m_bt_scan_results = g_Platform->getBluetoothScanResults();
386
387 m_src_choice->Clear();
388 m_src_choice->Append(m_bt_scan_results[0]); // scan status
389
390 unsigned i = 1;
391 while ((i + 1) < m_bt_scan_results.GetCount()) {
392 wxString item1 = m_bt_scan_results[i] + ";";
393 wxString item2 = m_bt_scan_results.Item(i + 1);
394 m_src_choice->Append(item1 + item2);
395 i += 2;
396 }
397
398 if (m_bt_scan_results.GetCount() > 1) {
399 m_src_choice->SetSelection(1);
400 }
401
402 // Watch for changes. When no changes occur after n seconds, stop the
403 // scan
404 if (m_bt_no_change_counter > 5) StopScan();
405 int count = static_cast<int>(m_bt_scan_results.GetCount());
406 if (count == m_bt_last_result_count)
407 m_bt_no_change_counter++;
408 else
409 m_bt_no_change_counter = 0;
410 m_bt_last_result_count = static_cast<int>(m_bt_scan_results.GetCount());
411
412 // Absolute fallback
413 if (m_bt_scanning >= 15) {
414 StopScan();
415 }
416 }
417
418 void StopScan() {
419 m_scan_timer.Stop();
420 g_Platform->stopBluetoothScan();
421 m_bt_scanning = 0;
422 if (m_scan_btn) {
423 m_scan_btn->SetLabel(_("BT Scan"));
424 m_scan_btn->Enable();
425 }
426 }

689
690 // Check available SocketCAN interfaces, empty list if not linux
691#if !defined(__linux__) || defined(__ANDROID__)
692 std::vector<std::string> GetAvailableSocketCanInterfaces() {
693 return {{}};
694 }
695#else
696 static int print_intf(const struct intf_entry* entry, void*) {
697 std::string iface = entry->intf_name;
698 if (entry->intf_type == 1 && iface.find("can") != std::string::npos) {
699 can_if_candidates.emplace_back(entry->intf_name);
700 }
701 return 0;
702 }
703
704 static std::vector<std::string> GetAvailableSocketCanInterfaces() {
705 std::vector<std::string> rv;
706 can_if_candidates.clear();
707
708 if ((intf = intf_open()) == nullptr) {
709 wxLogWarning("Error opening interface list");
710 return rv;
711 }
712 if (intf_loop(intf, print_intf, nullptr) < 0) {
713 wxLogWarning("Error looping over interface list");
714 }
715 intf_close(intf);
716

732 if (if_request.ifr_flags & IFF_UP) rv.push_back(iface);
733 }
