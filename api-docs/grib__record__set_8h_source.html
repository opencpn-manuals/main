166 GribRecordSet(unsigned int id) : m_Reference_Time(-1), m_ID(id) {
167 for (int i = 0; i < Idx_COUNT; i++) {
168 m_GribRecordPtrArray[i] = nullptr;
169 m_GribRecordUnref[i] = false;
170 }
171 }
172
173 virtual ~GribRecordSet() { RemoveGribRecords(); }
174
184 void SetUnRefGribRecord(int i, GribRecord *pGR) {
185 assert(i >= 0 && i < Idx_COUNT);
186 if (m_GribRecordUnref[i] == true) {
187 delete m_GribRecordPtrArray[i];
188 }
189 m_GribRecordPtrArray[i] = pGR;
190 m_GribRecordUnref[i] = true;
191 }
192
198 void RemoveGribRecords() {
199 for (int i = 0; i < Idx_COUNT; i++) {
200 if (m_GribRecordUnref[i] == true) {
201 delete m_GribRecordPtrArray[i];
202 }
203 }
204 }
205
208 time_t m_Reference_Time;
210 unsigned int m_ID;
211
220 GribRecord *m_GribRecordPtrArray[Idx_COUNT];
221
222private:
223 // grib records files are stored and owned by reader mapGribRecords
224 // interpolated grib are not, keep track of them
225 bool m_GribRecordUnref[Idx_COUNT];
226};
